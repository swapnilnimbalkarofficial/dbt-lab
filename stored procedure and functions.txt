18. to check if a number is even or odd using if then statement

delimiter //
create procedure checkEvenOdd(in p_number int)
begin	
	if mod(p_number, 2)=0 then
		select 'even';
	else
		select 'odd';
	end if;
end//

delimiter ;

call checkEvenOdd(3);


19. cal factorial of number using loop and leave

delimiter //
create procedure calFactorial(in no int, out fact bigint)
begin
	declare counter int default 1;
	declare factorial bigint default 1;
	myloop:loop
		set factorial=factorial*counter;
		set counter=counter+1;
		if(counter>no) then 
		leave myloop;
		leave myloop;
		end if;
	end loop;
end//
delimiter ;

call calFactorial(10);
